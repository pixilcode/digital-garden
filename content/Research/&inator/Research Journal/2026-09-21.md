## Function Pointers

Working example:

```c
int add(int *x, int *y) {
    return *x + *y;
}

int calc(int (*f)(int*, int*), int x, int y) {
    return (*f)(&x, &y);
}

int main() {
    int result = calc(&add, 1, 2);
    return 0;
}
```

Relevant LLVM output:

```llvm
; Function Attrs: nounwind uwtable
define dso_local i32 @add(ptr noundef %x, ptr noundef %y) #0 {
entry:
  %0 = load i32, ptr %x, align 4, !tbaa !4
  %1 = load i32, ptr %y, align 4, !tbaa !4
  %add = add nsw i32 %0, %1
  ret i32 %add
}

; Function Attrs: nounwind uwtable
define dso_local i32 @calc(ptr noundef %f, i32 noundef %x, i32 noundef %y) #0 {

  %y.addr = alloca i32, align 4
  store i32 %x, ptr %x.addr, align 4, !tbaa !4
```
